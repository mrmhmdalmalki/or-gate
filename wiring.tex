% ============================================================
%  OR gate - physical wiring diagram.
%  Three 2N3904 packages (flat face toward you, legs DOWN, E B C).
%  Stage 1 = NOR (Q1, Q2 in parallel, shared R_C1) gives A-bar+B;
%  Stage 2 = NOT (Q3) inverts it back to A+B at the output.
%  Drawn the way the parts physically sit on the board.
%  Compile:  pdflatex wiring.tex
% ============================================================
\documentclass[border=12pt]{standalone}
\usepackage{circuitikz}
\usepackage{tikz}
\usetikzlibrary{calc}

\def\w{0.95}
% TO-92 package: #1 = x-centre, #2 = reference name
\newcommand{\pkg}[2]{%
  \begin{scope}[shift={(#1,0)}]
    \draw[fill=black!10] (-\w,4.2) -- (-\w,5.2) arc (180:0:\w) -- (\w,4.2) -- cycle;
    \node at (0,4.6) {\small\textbf{2N3904}};
    \node[font=\footnotesize] at (0,5.55) {#2};
    \foreach \dx in {-0.6,0,0.6}{ \draw (\dx,4.2) -- (\dx,3.5); }
    \node[red,font=\bfseries\small,left=1pt]  at (-0.6,3.85) {E};
    \node[red,font=\bfseries\small,right=1pt] at ( 0.0,3.85) {B};
    \node[red,font=\bfseries\small,right=1pt] at ( 0.6,3.85) {C};
  \end{scope}}

% ground symbol (pic): same stroke as everything, just extra gap between bars
\tikzset{gnd/.pic={
  \draw
    (0,0) -- (0,-0.18)
    (-0.34,-0.18) -- (0.34,-0.18)
    (-0.21,-0.37) -- (0.21,-0.37)
    (-0.08,-0.56) -- (0.08,-0.56);
}}
\begin{document}
\begin{tikzpicture}[line width=0.8pt, font=\sffamily]
  \ctikzset{tripoles/thickness=1, bipoles/thickness=1}

  % ---------- power rails ----------
  \draw (-2,6.4) -- (14.6,6.4);
  \node[anchor=west] at (5.6,6.65) {$+5\,\mathrm{V}$ (Vcc)};
  \draw (-2,0) -- (14.6,0);
  \draw (-1.4,0) -- (-1.4,-0.05) pic{gnd};
  \node[anchor=west] at (-1.1,-0.45) {GND (0 V)};

  % ---------- transistors ----------
  \pkg{1.5}{Q1}
  \pkg{6.5}{Q2}
  \pkg{11.5}{Q3}

  % ===== Stage 1 : NOR (Q1, Q2 parallel) =====
  % emitters -> GND
  \draw (0.9,3.5) -- (0.9,0);  \fill (0.9,0) circle (1.6pt);
  \draw (5.9,3.5) -- (5.9,0);  \fill (5.9,0) circle (1.6pt);
  % bases -> R_B -> inputs
  \draw (1.5,3.5) -- (1.5,2.9) to[R] (1.5,1.1)
        -- (1.5,0.6) node[ocirc]{} node[right]{\,Input $A$};
  \node[anchor=east] at (0.55,2.0) {$R_{B1}=10\,\mathrm{k}\Omega$};
  \draw (6.5,3.5) -- (6.5,2.9) to[R] (6.5,1.1)
        -- (6.5,0.6) node[ocirc]{} node[right]{\,Input $B$};
  \node[anchor=east] at (5.6,2.0) {$R_{B2}=10\,\mathrm{k}\Omega$};
  % collector bus + shared R_C1 -> +5 V
  \draw (2.1,3.5) -- (2.1,3.0) -- (7.1,3.0) -- (7.1,3.5);
  \fill (3.5,3.0) circle (1.6pt);
  \draw (3.5,3.0) to[R, l_={$R_{C1}=1\,\mathrm{k}\Omega$}] (3.5,6.4);
  \fill (3.5,6.4) circle (1.6pt);
  \fill (7.1,3.0) circle (1.6pt);

  % ===== interstage: NOR node (A-bar+B) -> R_B3 -> Q3 base =====
  \draw (7.1,3.0) -- (7.1,2.0) -- (8.2,2.0)
        to[R, l_={$R_{B3}=10\,\mathrm{k}\Omega$}] (9.7,2.0) -- (11.5,2.0) -- (11.5,3.5);
  \node[anchor=west,font=\footnotesize] at (7.25,2.5) {$\overline{A+B}$};

  % ===== Stage 2 : NOT (Q3) =====
  % emitter -> GND
  \draw (10.9,3.5) -- (10.9,0);  \fill (10.9,0) circle (1.6pt);
  % collector -> R_C3 -> +5 V, and Output
  \draw (12.1,3.5) -- (12.1,3.0) -- (13.0,3.0);
  \fill (13.0,3.0) circle (1.6pt);
  \draw (13.0,3.0) to[R, l_={$R_{C3}=1\,\mathrm{k}\Omega$}] (13.0,6.4);
  \fill (13.0,6.4) circle (1.6pt);
  \draw (13.0,3.0) -- (14.3,3.0) node[ocirc]{} node[right]{Output $Y=A+B$};

\end{tikzpicture}
\end{document}
