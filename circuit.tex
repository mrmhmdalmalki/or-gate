% ============================================================
%  OR gate - logic diagram.
%  OR = a NOR gate followed by a NOT gate:
%        Y = NOT(NOT(A+B)) = A+B
%  Both sub-gates already exist in this project (transistor-level):
%  the NOR board and the NOT board.  Nothing new to build - just
%  wire a NOR board's output into a NOT board.
%  Compile:  pdflatex circuit.tex
% ============================================================
\documentclass[border=16pt]{standalone}
\usepackage{circuitikz}
\usepackage{amsmath}
\usetikzlibrary{calc}

\begin{document}
\begin{circuitikz}[line width=1.1pt, scale=1.5, transform shape]
  \ctikzset{logic ports=american, logic ports/thickness=1}

  \node[nor port] (g1) at (0,0)   {};
  \node[not port] (g2) at (3.8,0) {};
  \node[below=4pt,font=\footnotesize] at (g1.south) {NOR};
  \node[below=4pt,font=\footnotesize] at (g2.south) {NOT};

  % inputs A, B
  \draw (g1.in 1) -- ++(-1.5,0) node[ocirc]{} node[left]{$A$};
  \draw (g1.in 2) -- ++(-1.5,0) node[ocirc]{} node[left]{$B$};

  % NOR output -> NOT input
  \draw (g1.out) -- (g2.in);
  \node[above=1pt,font=\footnotesize] at ($(g1.out)!0.5!(g2.in)$) {$\overline{A+B}$};

  % output
  \draw (g2.out) -- ++(1.6,0) node[ocirc]{} node[right]{Output $Y=A+B$};

\end{circuitikz}
\end{document}
